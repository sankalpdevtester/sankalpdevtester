1. **`src/blockchain.py`** - Main blockchain class
```python
import hashlib
import json
from time import time
from typing import List, Dict
from src.block import Block
from src.transaction import Transaction
from src.proof_of_work import ProofOfWork

class Blockchain:
    def __init__(self):
        self.chain: List[Block] = []
        self.pending_transactions: List[Transaction] = []
        self.difficulty = 4
        self.mining_reward = 50
        self.create_genesis_block()
    
    def create_genesis_block(self):
        """Create the first block"""
        genesis = Block(0, [], time(), "0")
        genesis.hash = self.calculate_hash(genesis)
        self.chain.append(genesis)
    
    def get_latest_block(self) -> Block:
        return self.chain[-1]
    
    def add_transaction(self, transaction: Transaction) -> bool:
        """Add transaction to pending pool"""
        if not transaction.is_valid():
            return False
        self.pending_transactions.append(transaction)
        return True
    
    def mine_pending_transactions(self, miner_address: str):
        """Mine all pending transactions"""
        reward_tx = Transaction("system", miner_address, self.mining_reward)
        self.pending_transactions.append(reward_tx)
        
        block = Block(
            len(self.chain),
            self.pending_transactions,
            time(),
            self.get_latest_block().hash
        )
        
        pow = ProofOfWork(block, self.difficulty)
        nonce, block_hash = pow.mine()
        
        block.nonce = nonce
        block.hash = block_hash
        
        self.chain.append(block)
        self.pending_transactions = []
    
    def get_balance(self, address: str) -> float:
        """Calculate balance for an address"""
        balance = 0
        for block in self.chain:
            for tx in block.transactions:
                if tx.from_address == address:
                    balance -= tx.amount
                if tx.to_address == address:
                    balance += tx.amount
        return balance
    
    def is_chain_valid(self) -> bool:
        """Validate entire blockchain"""
        for i in range(1, len(self.chain)):
            current = self.chain[i]
            previous = self.chain[i-1]
            
            if current.hash != self.calculate_hash(current):
                return False
            if current.previous_hash != previous.hash:
                return False
        return True
    
    @staticmethod
    def calculate_hash(block: Block) -> str:
        """Calculate SHA256 hash of block"""
        block_string = json.dumps({
            'index': block.index,
            'transactions': [tx.__dict__ for tx in block.transactions],
            'timestamp': block.timestamp,
            'previous_hash': block.previous_hash,
            'nonce': block.nonce
        }, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()
```

3. **`src/transaction.py`**
```python
import hashlib
import json
from src.wallet import Wallet

class Transaction:
    def __init__(self, from_address: str, to_address: str, amount: float):
        self.from_address = from_address
        self.to_address = to_address
        self.amount = amount
        self.timestamp = time.time()
        self.signature = None
    
    def calculate_hash(self) -> str:
        tx_string = json.dumps({
            'from': self.from_address,
            'to': self.to_address,
            'amount': self.amount,
            'timestamp': self.timestamp
        }, sort_keys=True)
        return hashlib.sha256(tx_string.encode()).hexdigest()
    
    def sign_transaction(self, wallet: Wallet):
        """Sign transaction with sender's private key"""
        if self.from_address != wallet.public_key:
            raise Exception("Cannot sign transaction from different wallet")
        self.signature = wallet.sign(self.calculate_hash())
    
    def is_valid(self) -> bool:
        """Verify transaction signature"""
        if self.from_address == "system":
            return True
        if not self.signature:
            return False
        return Wallet.verify(self.from_address, self.calculate_hash(), self.signature)
    
    def to_dict(self):
        return {
            'from': self.from_address,
            'to': self.to_address,
            'amount': self.amount,
            'timestamp': self.timestamp
```

```python
from ecdsa import SigningKey, VerifyingKey, SECP256k1
import hashlib

class Wallet:
    def __init__(self):
        self.private_key = SigningKey.generate(curve=SECP256k1)
        self.public_key = self.private_key.get_verifying_key().to_string().hex()
    
    def sign(self, message: str) -> str:
        """Sign a message with private key"""
        signature = self.private_key.sign(message.encode())
        return signature.hex()
    
    @staticmethod
    def verify(public_key: str, message: str, signature: str) -> bool:
        """Verify signature with public key"""
        try:
            vk = VerifyingKey.from_string(bytes.fromhex(public_key), curve=SECP256k1)
            vk.verify(bytes.fromhex(signature), message.encode())
            return True
        except:
            return False
```

5. **`src/proof_of_work.py`**
```python
import hashlib
import json

class ProofOfWork:
    def __init__(self, block, difficulty: int):
        self.block = block
        self.difficulty = difficulty
        self.target = "0" * difficulty
    
    def mine(self) -> tuple:
        """Find nonce that produces hash with required difficulty"""
        nonce = 0
        while True:
            block_hash = self.calculate_hash(nonce)
            if block_hash[:self.difficulty] == self.target:
                return nonce, block_hash
            nonce += 1
    
    def calculate_hash(self, nonce: int) -> str:
        block_string = json.dumps({
            'index': self.block.index,
            'transactions': [tx.__dict__ for tx in self.block.transactions],
            'timestamp': self.block.timestamp,
            'previous_hash': self.block.previous_hash,
            'nonce': nonce
        }, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()
```
